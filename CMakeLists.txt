cmake_minimum_required(VERSION 3.16)
project(TelemetryGateway VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/boost_1_85_0/boost/asio.hpp")
 set(Boost_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/boost_1_85_0")
else()
 find_package(Boost 1.75 REQUIRED)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite-amalgamation-3460000/sqlite3.c")
 add_library(sqlite_vendor STATIC third_party/sqlite-amalgamation-3460000/sqlite3.c)
 target_include_directories(sqlite_vendor PUBLIC third_party/sqlite-amalgamation-3460000)
 target_link_libraries(sqlite_vendor PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
 add_library(SQLite::SQLite3 ALIAS sqlite_vendor)
else()
 find_package(SQLite3 REQUIRED)
endif()
add_library(telemetry_core src/core.cpp)
target_include_directories(telemetry_core PUBLIC include ${Boost_INCLUDE_DIRS})
target_link_libraries(telemetry_core PUBLIC SQLite::SQLite3 Threads::Threads)
target_compile_definitions(telemetry_core PUBLIC BOOST_ERROR_CODE_HEADER_ONLY)
if(WIN32)
 target_link_libraries(telemetry_core PUBLIC ws2_32 mswsock)
 target_compile_definitions(telemetry_core PUBLIC _WIN32_WINNT=0x0601 NOMINMAX)
endif()
add_executable(telemetry_server src/server.cpp src/network.cpp)
target_link_libraries(telemetry_server PRIVATE telemetry_core)
add_executable(device_simulator src/simulator.cpp)
target_link_libraries(device_simulator PRIVATE telemetry_core)
if(MSVC)
 target_compile_options(telemetry_core PUBLIC /W4 /utf-8)
else()
 target_compile_options(telemetry_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE telemetry_core)
add_test(NAME core_tests COMMAND core_tests)
